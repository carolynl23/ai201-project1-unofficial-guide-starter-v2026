"""Two candidate gate signals, measured on all 18 questions.

A: gap between rank 1 and rank 2 — is an uncovered near-corpus question 'flat'
   (many template docs equally close) where a covered one is 'peaked'?
B: a capitalised proper noun in the question that appears nowhere in the corpus.
"""
import re
import config, questions as qs
from store import search
from ingest import load_documents
from check_boundary import PROBES

docs = load_documents()
vocab = set()
for d in docs:
    vocab |= set(re.findall(r"[a-z0-9']+", d.text.lower()))

def ghost_nouns(q):
    """Capitalised words, not sentence-initial, absent from the corpus."""
    tokens = re.findall(r"\b[A-Za-z][a-z']+\b", q)
    return [w for w in tokens[1:] if w[0].isupper() and w.lower() not in vocab]

groups = (
    [(item["question"], True) for item in qs.answered()]
    + [(q, lab == "covered") for q, lab in PROBES]
    + [(q, False) for q in qs.OUT_OF_SCOPE]
)

print(f"{'covered':<8} {'d1':>6} {'d2':>6} {'gap':>6}  {'ghosts':<22} question")
rows = []
for q, covered in groups:
    r = search(q, top_k=5, corpus=config.CORPUS)
    d1, d2 = r[0].distance, r[1].distance
    g = ghost_nouns(q)
    rows.append((covered, d1, d2 - d1, g, q))
    print(f"{str(covered):<8} {d1:>6.3f} {d2:>6.3f} {d2-d1:>6.3f}  {str(g):<22} {q[:44]}")

cov = [r for r in rows if r[0]]
unc = [r for r in rows if not r[0]]
print(f"\nSignal A (gap): covered min gap {min(r[2] for r in cov):.3f}; "
      f"uncovered near-corpus gaps {[round(r[2],3) for r in unc if r[1] < 0.6]}")
print(f"Signal B (ghost nouns): covered with ghosts "
      f"{[r[4][:30] for r in cov if r[3]]}; "
      f"uncovered-under-cutoff with ghosts {[r[4][:30] for r in unc if r[3] and r[1] < 0.6]}")

"""Variant B2: a capitalised entity in the question that appears in none of the
chunks the gate is about to accept. Cheaper than a corpus scan, and it also
catches retrieval bringing back the wrong building for a real one."""
import re
import config, gate, questions as qs
from store import search
from check_boundary import PROBES

def entities(q):
    tokens = re.findall(r"\b[A-Za-z][A-Za-z']+\b", q)
    return [w for w in tokens[1:] if w[0].isupper()]

groups = (
    [(item["question"], True) for item in qs.answered()]
    + [(q, lab == "covered") for q, lab in PROBES]
    + [(q, False) for q in qs.OUT_OF_SCOPE]
)

print(f"{'covered':<8} {'d1':>6}  {'entities':<26} {'missing from grounding':<24} verdict")
fp = fn = 0
for q, covered in groups:
    r = search(q, top_k=5, corpus=config.CORPUS)
    kept = gate.relevant(r, threshold=config.THRESHOLD)
    blob = " ".join(c.text for c in kept).lower()
    ents = entities(q)
    missing = [e for e in ents if e.lower() not in blob]
    would_refuse = bool(missing)
    if covered and would_refuse:
        fp += 1; verdict = "FALSE REFUSAL"
    elif not covered and would_refuse:
        verdict = "refuses (good)"
    else:
        verdict = ""
    print(f"{str(covered):<8} {r[0].distance:>6.3f}  {str(ents):<26} {str(missing):<24} {verdict}")

print(f"\nfalse refusals of covered questions: {fp}")
